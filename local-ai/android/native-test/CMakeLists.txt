cmake_minimum_required(VERSION 3.22)
project(hosttest C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
foreach(o LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_BUILD_APP LLAMA_OPENSSL LLAMA_SUBPROCESS)
  set(${o} OFF CACHE BOOL "" FORCE)
endforeach()
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
add_subdirectory(${LLAMA_SRC} build-llama)
find_package(JNI REQUIRED)
add_library(localai SHARED ${JNI_SRC})
target_include_directories(localai PRIVATE ${CMAKE_SOURCE_DIR}/stub ${JNI_INCLUDE_DIRS} ${LLAMA_SRC}/include ${LLAMA_SRC}/common ${LLAMA_SRC}/ggml/include ${LLAMA_SRC}/tools/mtmd)
target_link_libraries(localai llama llama-common mtmd ggml)
